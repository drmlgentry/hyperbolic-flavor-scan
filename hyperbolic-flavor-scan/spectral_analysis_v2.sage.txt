#!/usr/bin/env sage
# spectral_analysis_v2.sage (final – fixed JSON serialization)

import snappy
import json
from sage.all import *
from math import sinh, sin

# ------------------------------------------------------------
# JSON serialization – handles Sage numeric types without isinstance(type)
# ------------------------------------------------------------
def to_json(obj):
    # Convert to int if possible
    try:
        if obj in ZZ:
            return int(obj)
    except:
        pass
    # Convert to float if possible (works for Sage reals)
    try:
        f = float(obj)
        # If it's an integer float, return int
        if f.is_integer():
            return int(f)
        return f
    except:
        pass
    # Complex numbers
    try:
        if obj in CC:
            return {'real': float(obj.real()), 'imag': float(obj.imag())}
    except:
        pass
    # Lists and tuples
    if isinstance(obj, (list, tuple)):
        return [to_json(x) for x in obj]
    # Dictionaries
    if isinstance(obj, dict):
        return {to_json(k): to_json(v) for k, v in obj.items()}
    # Fallback to string
    return str(obj)

# ------------------------------------------------------------
# Geodesic lengths (distinct)
# ------------------------------------------------------------
def get_lengths(M, cutoff=4.5):
    try:
        ls = M.length_spectrum(cutoff=cutoff)
        lens = []
        for g in ls:
            try:
                v = g.length
                if callable(v):
                    v = v()
                ell = float(complex(v).real)
                lens.append(ell)
            except:
                continue
        return sorted(set(lens))
    except Exception as e:
        print(f"  Error: {e}")
        return []

# ------------------------------------------------------------
# 1. Hecke operator spectrum
# ------------------------------------------------------------
def hecke_spectrum():
    res = {}
    # Brandt module discriminant 3
    try:
        B = QuaternionAlgebra(QQ, -1, -3)
        disc = B.discriminant()
        BM = BrandtModule(disc)
        dim = BM.dimension()
        res['brandt_disc3'] = {'dimension': int(dim)}
        if dim > 0:
            hecke = {}
            for p in primes(2, 50):
                if p % 3 == 1:   # splits in Q(√-3)
                    Tp = BM.hecke_matrix(p)
                    if dim == 1:
                        hecke[int(p)] = int(Tp[0,0])
                    else:
                        hecke[int(p)] = [float(x) for x in Tp.eigenvalues()]
            res['brandt_disc3']['hecke'] = hecke
    except Exception as e:
        res['brandt_disc3'] = {'error': str(e)}
    # Newform level 5 weight 2
    try:
        forms = Newforms(5, 2, names='a')
        if forms:
            f = forms[0]
            coeffs = f.qexp(51)
            hecke = {}
            for p in primes(2, 50):
                if p != 5:
                    ap = coeffs[p]
                    if ap in ZZ:
                        hecke[int(p)] = int(ap)
                    else:
                        hecke[int(p)] = float(ap)
            res['newform_level5'] = {'dim': 1, 'hecke': hecke}
        else:
            res['newform_level5'] = {'error': 'No newform'}
    except Exception as e:
        res['newform_level5'] = {'error': str(e)}
    return res

# ------------------------------------------------------------
# 2. Selberg zeros via sine sum (approximate)
# ------------------------------------------------------------
def selberg_zeros(lengths, max_zeros=10):
    if not lengths:
        return []
    t_vals = [t/100.0 for t in range(2001)]  # 0 → 20, step 0.01
    F = [0.0]*len(t_vals)
    for ell in lengths:
        if ell < 1e-6:
            continue
        w = ell / (2 * sinh(ell/2))
        for i, t in enumerate(t_vals):
            F[i] += w * sin(t * ell)
    zeros = []
    for i in range(1, len(F)):
        if F[i-1] * F[i] < 0:
            t0, t1 = t_vals[i-1], t_vals[i]
            z = t0 + (t1 - t0) * (-F[i-1]) / (F[i] - F[i-1])
            zeros.append(z)
    zeros = sorted(set([round(z, 5) for z in zeros if z > 0]))
    return zeros[:max_zeros]

# ------------------------------------------------------------
# 3. Laplace eigenvalues (λ = 1/4 + r²)
# ------------------------------------------------------------
def laplace_from_zeros(zeros):
    return [0.25 + r*r for r in zeros]

# ------------------------------------------------------------
# 4. Transfer resonances (s = 0.5 + i r)
# ------------------------------------------------------------
def transfer_resonances(zeros):
    return [0.5 + 1j*z for z in zeros]

# ------------------------------------------------------------
# Main
# ------------------------------------------------------------
manifolds = [
    (1, 'PMNS', snappy.OrientableClosedCensus[1]),
    (43, 'CKM', snappy.OrientableClosedCensus[43])
]

results = {'hecke': hecke_spectrum()}

for idx, name, M in manifolds:
    print(f"\nProcessing {name} (idx {idx})...")
    vol = float(M.volume())
    lens = get_lengths(M, cutoff=4.5)
    print(f"  Geodesics ≤4.5: {len(lens)}")

    zeros = selberg_zeros(lens, max_zeros=10)
    laplace = laplace_from_zeros(zeros)
    resonances = transfer_resonances(zeros)

    print(f"  Selberg zeros (r): {zeros[:3]}")
    print(f"  Laplace λ: {[round(l,4) for l in laplace[:3]]}")

    results[name] = {
        'index': idx,
        'volume': vol,
        'num_geodesics': len(lens),
        'selberg_zeros': zeros,
        'laplace_eigenvalues': laplace,
        'transfer_resonances': [{'real': s.real, 'imag': s.imag} for s in resonances]
    }

# Save JSON
outfile = 'spectral_final.json'
with open(outfile, 'w') as f:
    json.dump(to_json(results), f, indent=2)
print(f"\nResults saved to {outfile}")

# Summary
print("\n" + "="*60)
print("SUMMARY")
print("="*60)
for name in ['PMNS', 'CKM']:
    if name in results:
        d = results[name]
        print(f"\n{name}:")
        print(f"  Volume: {d['volume']:.6f}")
        print(f"  Geodesics: {d['num_geodesics']}")
        print(f"  First 3 Selberg zeros: {d['selberg_zeros'][:3]}")
        print(f"  First 3 Laplace λ: {[round(l,4) for l in d['laplace_eigenvalues'][:3]]}")
        if d['transfer_resonances']:
            s0 = d['transfer_resonances'][0]
            print(f"  First resonance: s = {s0['real']} + {s0['imag']:.5f}i")
print("\nHecke spectrum saved inside the JSON file.")
